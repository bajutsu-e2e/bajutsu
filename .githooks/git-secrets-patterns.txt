# Additional git-secrets prohibited patterns beyond `git secrets --register-aws`.
# One extended-regex pattern per line; a leading `#` line is a comment. Registered into this
# clone's local git config by `make hooks` -> scripts/git-secrets-setup.sh. Keep this list narrow
# and specific to secrets this repo actually handles (see .env.example / SECURITY.md) — a broad
# pattern trades false positives for coverage it doesn't need.

# Anthropic API key / OAuth token (.env.example: ANTHROPIC_API_KEY, CLAUDE_CODE_OAUTH_TOKEN).
sk-ant-[A-Za-z0-9_-]{20,}

# A PEM private-key block pasted into a file or commit. No leading `-----`: a pattern starting
# with a hyphen gets misread as a command-line option (by `git secrets --add` and by the `grep`
# it shells out to), and the trailing dashes alone are already unambiguous.
BEGIN[A-Z ]*PRIVATE KEY-----
